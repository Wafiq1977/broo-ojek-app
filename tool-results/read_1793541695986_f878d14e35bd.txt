     2→

   101→        if (!res.ok || !data.ok) throw new Error(data.error || "Gagal membuat pesanan");

   105→            name: o.driverName,
   106→            plate: "B 4521 KXP",

   133→            service: selectedService,
   134→            pickup,
   135→            destination,
   136→            km: effectiveKm,
   137→            scheduledAt,
   138→          }),

   181→          <motion.div
   182→            initial={{ y: "100%" }}
   183→            animate={{ y: 0 }}
   184→            exit={{ y: "100%" }}

   299→                        className="grid grid-cols-2 gap-2"

   309→                        </div>

   311→                          <label className="mb-1 block text-[11px] font-semibold text-foreground/70">Jam</label>

   314→                            value={bookingTime}
   315→                            onChange={(e) => setBookingTime(e.target.value)}

   318→                        </div>

   321→
   322→                    {/* quick pick destinations */}

   326→                          <BrooIcon name="MapPin" className="h-3.5 w-3.5 text-primary" />
   327→                          Pilih cepat

   329→                        {SAVED_PLACES.slice(0, 4).map((p) => (
   330→                          <button

   332→                            onClick={() => { setDestination(`${p.name} — ${p.address}`); setKm(p.km); }}
   333→                            className="flex w-full items-center gap-3 rounded-2xl border border-border bg-card p-3 text-left tap-scale"
   334→                          >
   335→                            <BrooIcon name="MapPin" className="h-4 w-4 text-primary" />
   336→                            <div className="min-w-0 flex-1">
   337→                              <p className="truncate text-sm font-semibold">{p.name}</p>
   338→                              <p className="truncate text-[11px] text-muted-foreground">{p.address}</p>
   339→                            </div>
   340→                            <span className="text-[11px] font-bold text-foreground">{p.km} km</span>
   341→                          </button>

   343→                      </div>

   346→                    <button

   353→                    </button>
   354→                  </div>
   355→                </motion.div>

   358→              {/* ============ STEP 2: SERVICE SELECTION (Grab-like) ============ */}

   368→                    <div className="mx-auto mb-3 h-1.5 w-10 rounded-full bg-white/40" />

   370→                      <button
   371→                        onClick={() => setStage("location")}

   373→                        aria-label="Kembali"
   374→                      >
   375→                        <BrooIcon name="ChevronLeft" className="h-5 w-5" />
